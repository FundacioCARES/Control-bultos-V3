// server.js // Backend que sustituye a los 8 flujos de Power Automate con disparador // HTTP de la app "Control de Bultos". Cada función de aquí abajo hace // exactamente lo mismo que hacía su flujo equivalente. require('dotenv').config(); const express = require('express'); const crypto = require('crypto'); const { listItems, createItem, updateItem } = require('./graphClient'); const app = express(); app.use(express.json()); // ============================================================ // NOMBRES INTERNOS DE COLUMNA // ------------------------------------------------------------ // SharePoint a veces usa, por dentro, un nombre distinto al que ves en // pantalla (sobre todo si el nombre visible tiene tildes o símbolos). // Confirmados: Codigo (C_x00f3_digo), ClaveUnica (Clave), Manipulacion y los // de ListadoOperadores. El resto se han puesto igual que el nombre visible; // si alguno no coincidiera, el servidor dará error 500 y en su registro // aparecerá el nombre de la columna que falla (ver README, apartado // "Cómo comprobar el nombre interno de una columna"). // ============================================================ const COL = { registroBultos: { codigo: 'C_x00f3_digo', // columna "Codigo" claveUnica: 'Clave', // columna "ClaveUnica" operario: 'Operario', manipulacion: 'Manipulacion', prendas: 'Prendas', fecha: 'Fecha', fechaHora: 'FechaHora', cajaMedias: 'CajaMedias', observaciones: 'Observaciones' // columna nueva (texto) para el tipo de reoperado }, manipulaciones: { codigo: 'Codigo', recordatorio: 'Recordatorio', ordenNumero: 'OrdenNumero' }, sesiones: { operario: 'Operario', manipulacion: 'Manipulacion', inicio: 'Inicio', fin: 'Fin' }, operadores: { operario: 'Title', // primera columna de la lista, siempre "Title" por dentro nombre: 'field_1', // columna "NOMBRE" contrato: 'field_2' // columna "CONTRATO" } }; const LISTA = { registroBultos: process.env.LISTA_REGISTRO_BULTOS, manipulaciones: process.env.LISTA_MANIPULACIONES, sesiones: process.env.LISTA_SESIONES, operadores: process.env.LISTA_OPERADORES }; // ============================================================ // SEGURIDAD: CORS + clave de dispositivo // ============================================================ app.use((req, res, next) => { res.setHeader('Access-Control-Allow-Origin', process.env.ALLOWED_ORIGIN || ''); res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type, X-Device-Key'); if (req.method === 'OPTIONS') return res.sendStatus(204); next(); }); app.use((req, res, next) => { const clave = req.header('X-Device-Key'); if (!clave || clave !== process.env.DEVICE_KEY) { return res.status(401).json({ error: 'Clave de dispositivo no válida.' }); } next(); }); // ============================================================ // VALIDACIONES (las mismas reglas que ya tiene la app) // ============================================================ function codigoBultoValido(codigo) { return typeof codigo === 'string' && /^0038\d{16}$/.test(codigo); } function textoValido(valor, longitudMax = 100) { return typeof valor === 'string' && valor.trim().length > 0 && valor.length <= longitudMax; } function prendasValidas(valor) { const n = parseInt(valor, 10); return Number.isInteger(n) && n > 0 && n <= 100000; } // Envuelve cada ruta para no repetir el try/catch en las 8 funciones. function ruta(fn) { return async (req, res) => { try { await fn(req, res); } catch (error) { console.error(error); res.status(500).json({ error: 'Error interno del servidor.' }); } }; } // ============================================================ // 1) GuardarBulto // ============================================================ app.post('/guardar-bulto', ruta(async (req, res) => { const { operario, codigo, manipulacion, prendas, cajaMedias, reoperado, observaciones } = req.body; // REOPERADO: prendas sueltas sin etiqueta de bulto -> fila sin código const esReoperado = reoperado === true; if (esReoperado && !(typeof manipulacion === 'string' && manipulacion.trim().toUpperCase() === 'REOPERADO')) { return res.status(400).json({ error: 'Manipulación no válida para reoperado.' }); } if (esReoperado && !textoValido(observaciones, 200)) { return res.status(400).json({ error: 'Indica el tipo de reoperado.' }); } if (!esReoperado && !codigoBultoValido(codigo)) return res.status(400).json({ error: 'Código de bulto no válido.' }); if (!textoValido(operario)) return res.status(400).json({ error: 'Operario no válido.' }); if (!textoValido(manipulacion)) return res.status(400).json({ error: 'Manipulación no válida.' }); if (!prendasValidas(prendas)) return res.status(400).json({ error: 'Número de prendas no válido.' }); const esCajaMedias = !esReoperado && cajaMedias === true; const ahora = new Date().toISOString(); if (!esCajaMedias && !esReoperado) { const existentes = await listItems(LISTA.registroBultos, { filtro: `fields/${COL.registroBultos.codigo} eq '${codigo}'` }); if (existentes.length > 0) { return res.status(409).json({ error: 'Este bulto ya ha sido registrado.' }); } } const claveUnica = (esCajaMedias || esReoperado) ? crypto.randomUUID() : codigo; await createItem(LISTA.registroBultos, { [COL.registroBultos.codigo]: esReoperado ? '' : codigo, [COL.registroBultos.operario]: operario, [COL.registroBultos.manipulacion]: manipulacion, [COL.registroBultos.prendas]: parseInt(prendas, 10), [COL.registroBultos.fecha]: ahora, [COL.registroBultos.fechaHora]: ahora, [COL.registroBultos.cajaMedias]: esCajaMedias, [COL.registroBultos.claveUnica]: claveUnica, ...(esReoperado ? { [COL.registroBultos.observaciones]: observaciones.trim() } : {}) }); res.status(200).json({ ok: true }); })); // ============================================================ // 2) EditarPrendas // ============================================================ app.post('/editar-prendas', ruta(async (req, res) => { const { operario, codigo, prendas, manipulacion } = req.body; if (!codigoBultoValido(codigo)) return res.status(400).json({ error: 'Código de bulto no válido.' }); if (!prendasValidas(prendas)) return res.status(400).json({ error: 'Número de prendas no válido.' }); let filtro = `fields/${COL.registroBultos.codigo} eq '${codigo}'`; if (textoValido(operario)) { filtro += ` and fields/${COL.registroBultos.operario} eq '${operario.replace(/'/g, "''")}'`; } const filas = await listItems(LISTA.registroBultos, { filtro }); if (filas.length === 0) return res.status(404).json({ error: 'Bulto no encontrado.' }); const campos = { [COL.registroBultos.prendas]: parseInt(prendas, 10) }; if (textoValido(manipulacion)) { campos[COL.registroBultos.manipulacion] = manipulacion; } await updateItem(LISTA.registroBultos, filas[0].id, campos); res.status(200).json({ ok: true }); })); // ============================================================ // 3) ObtenerManipulaciones // ============================================================ app.post('/manipulaciones', ruta(async (req, res) => { const filas = await listItems(LISTA.manipulaciones, { orden: `fields/${COL.manipulaciones.ordenNumero} asc` }); const resultado = filas.map(f => ({ codigo: f.fields[COL.manipulaciones.codigo], recordatorio: f.fields[COL.manipulaciones.recordatorio] || '' })); res.status(200).json(resultado); })); // ============================================================ // 4) ComprobarBulto // ============================================================ app.post('/comprobar-bulto', ruta(async (req, res) => { const { codigo } = req.body; if (!codigoBultoValido(codigo)) return res.status(400).json({ error: 'Código de bulto no válido.' }); const filas = await listItems(LISTA.registroBultos, { filtro: `fields/${COL.registroBultos.codigo} eq '${codigo}'` }); res.status(200).json({ existe: filas.length > 0 }); })); // ============================================================ // 5) IniciarSesion // ============================================================ app.post('/iniciar-sesion', ruta(async (req, res) => { const { operario, manipulacion } = req.body; if (!textoValido(operario) || !textoValido(manipulacion)) { return res.status(400).json({ error: 'Datos de sesión no válidos.' }); } const nueva = await createItem(LISTA.sesiones, { [COL.sesiones.operario]: operario, [COL.sesiones.manipulacion]: manipulacion, [COL.sesiones.inicio]: new Date().toISOString() }); res.status(200).json({ id: nueva.id }); })); // ============================================================ // 6) FinalizarSesion // ============================================================ app.post('/finalizar-sesion', ruta(async (req, res) => { const { id } = req.body; if (!textoValido(String(id || ''), 50)) return res.status(400).json({ error: 'Id de sesión no válido.' }); await updateItem(LISTA.sesiones, id, { [COL.sesiones.fin]: new Date().toISOString() }); res.status(200).json({ ok: true }); })); // ============================================================ // 7) ObtenerDetalleBulto // ============================================================ app.post('/detalle-bulto', ruta(async (req, res) => { const { codigo } = req.body; if (!codigoBultoValido(codigo)) return res.status(400).json({ error: 'Código de bulto no válido.' }); const filas = await listItems(LISTA.registroBultos, { filtro: `fields/${COL.registroBultos.codigo} eq '${codigo}'` }); if (filas.length === 0) return res.status(200).json({ encontrado: false }); const f = filas[0].fields; res.status(200).json({ encontrado: true, operario: f[COL.registroBultos.operario] || '', manipulacion: f[COL.registroBultos.manipulacion] || '', prendas: f[COL.registroBultos.prendas] || '', fecha: f[COL.registroBultos.fecha] || '' }); })); // ============================================================ // 8) ObtenerOperadores // ============================================================ app.post('/operarios', ruta(async (req, res) => { const filas = await listItems(LISTA.operadores, { orden: `fields/${COL.operadores.operario} asc` }); const resultado = filas.map(f => ({ operario: f.fields[COL.operadores.operario], nombre: f.fields[COL.operadores.nombre] || '' })); res.status(200).json(resultado); }));
